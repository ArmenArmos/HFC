      const WORKLET_CODE = `
class PcmCaptureProcessor extends AudioWorkletProcessor {
  constructor() {
    super();
    this.buffer = new Float32Array(2048);
    this.offset = 0;
  }
  process(inputs) {
    const input = inputs[0];
    if (!input || !input[0]) return true;
    const channel = input[0];
    for (let i = 0; i < channel.length; i++) {
      this.buffer[this.offset++] = channel[i];
      if (this.offset >= this.buffer.length) {
        this.port.postMessage(this.buffer.slice(0));
        this.offset = 0;
      }
    }
    return true;
  }
}
registerProcessor('pcm-capture-processor', PcmCaptureProcessor);
`;

      class LiveAudioManager {
        constructor(callbacks) {
          this.inputCtx = null;
          this.outputCtx = null;
          this.micStream = null;
          this.workletNode = null;
          this.fallbackProcessor = null;
          this.sourceNode = null;
          this.activeSources = new Set();
          this.nextStartTime = 0;
          this.onInputLevel = callbacks.onInputLevel;
          this.onOutputLevel = callbacks.onOutputLevel;
          this.onSpeakingStateChange = callbacks.onSpeakingStateChange;
        }

        processFloat32Frame(inputData, onPcmChunk) {
          let sum = 0;
          for (let i = 0; i < inputData.length; i++) {
            sum += inputData[i] * inputData[i];
          }
          const rms = Math.sqrt(sum / inputData.length);
          this.onInputLevel(Math.min(1, rms * 5));

          const pcm16 = new Int16Array(inputData.length);
          for (let i = 0; i < inputData.length; i++) {
            const s = Math.max(-1, Math.min(1, inputData[i]));
            pcm16[i] = s < 0 ? s * 0x8000 : s * 0x7fff;
          }

          const uint8 = new Uint8Array(pcm16.buffer);
          let binary = '';
          for (let i = 0; i < uint8.byteLength; i++) {
            binary += String.fromCharCode(uint8[i]);
          }
          onPcmChunk(btoa(binary));
        }

        async startMicrophone(onPcmChunk) {
          this.micStream = await navigator.mediaDevices.getUserMedia({
            audio: {
              sampleRate: 16000,
              channelCount: 1,
              echoCancellation: true,
              noiseSuppression: true,
              autoGainControl: true,
            },
          });

          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          this.inputCtx = new AudioCtx({ sampleRate: 16000 });
          this.outputCtx = new AudioCtx({ sampleRate: 24000 });

          if (this.inputCtx.state === 'suspended') await this.inputCtx.resume();
          if (this.outputCtx.state === 'suspended') await this.outputCtx.resume();

          this.sourceNode = this.inputCtx.createMediaStreamSource(this.micStream);

          try {
            const blob = new Blob([WORKLET_CODE], { type: 'application/javascript' });
            const workletUrl = URL.createObjectURL(blob);
            await this.inputCtx.audioWorklet.addModule(workletUrl);
            URL.revokeObjectURL(workletUrl);

            this.workletNode = new AudioWorkletNode(this.inputCtx, 'pcm-capture-processor');
            this.workletNode.port.onmessage = (event) => {
              this.processFloat32Frame(event.data, onPcmChunk);
            };
            this.sourceNode.connect(this.workletNode);
            this.workletNode.connect(this.inputCtx.destination);
          } catch {
            this.fallbackProcessor = this.inputCtx.createScriptProcessor(2048, 1, 1);
            this.fallbackProcessor.onaudioprocess = (event) => {
              const inputData = event.inputBuffer.getChannelData(0);
              this.processFloat32Frame(inputData, onPcmChunk);
            };
            this.sourceNode.connect(this.fallbackProcessor);
            this.fallbackProcessor.connect(this.inputCtx.destination);
          }
        }

        enqueueOutputPcm(base64Data) {
          if (!this.outputCtx) return;
          if (this.outputCtx.state === 'suspended') this.outputCtx.resume();

          const binaryString = atob(base64Data);
          const byteLength = binaryString.length;
          const bytes = new Uint8Array(byteLength);
          for (let i = 0; i < byteLength; i++) {
            bytes[i] = binaryString.charCodeAt(i);
          }

          const int16 = new Int16Array(bytes.buffer);
          const float32 = new Float32Array(int16.length);
          let sum = 0;
          for (let i = 0; i < int16.length; i++) {
            const sample = int16[i] / 32768.0;
            float32[i] = sample;
            sum += sample * sample;
          }

          const rms = Math.sqrt(sum / Math.max(1, float32.length));
          this.onOutputLevel(Math.min(1, rms * 4.5));

          const audioBuffer = this.outputCtx.createBuffer(1, float32.length, 24000);
          audioBuffer.getChannelData(0).set(float32);

          const source = this.outputCtx.createBufferSource();
          source.buffer = audioBuffer;
          source.connect(this.outputCtx.destination);

          const now = this.outputCtx.currentTime;
          if (this.nextStartTime < now) {
            this.nextStartTime = now + 0.02;
          }

          source.start(this.nextStartTime);
          this.nextStartTime += audioBuffer.duration;

          this.activeSources.add(source);
          this.onSpeakingStateChange(true);

          source.onended = () => {
            this.activeSources.delete(source);
            if (this.activeSources.size === 0) {
              this.onOutputLevel(0);
              this.onSpeakingStateChange(false);
            }
          };
        }

        interruptOutput() {
          this.activeSources.forEach((src) => {
            try {
              src.stop();
            } catch {}
          });
          this.activeSources.clear();
          if (this.outputCtx) {
            this.nextStartTime = this.outputCtx.currentTime;
          }
          this.onOutputLevel(0);
          this.onSpeakingStateChange(false);
        }

        stopAll() {
          this.interruptOutput();
          if (this.workletNode) {
            this.workletNode.port.onmessage = null;
            this.workletNode.disconnect();
            this.workletNode = null;
          }
          if (this.fallbackProcessor) {
            this.fallbackProcessor.disconnect();
            this.fallbackProcessor.onaudioprocess = null;
            this.fallbackProcessor = null;
          }
          if (this.sourceNode) {
            this.sourceNode.disconnect();
            this.sourceNode = null;
          }
          if (this.micStream) {
            this.micStream.getTracks().forEach((t) => t.stop());
            this.micStream = null;
          }
          if (this.inputCtx) {
            this.inputCtx.close().catch(() => {});
            this.inputCtx = null;
          }
          if (this.outputCtx) {
            this.outputCtx.close().catch(() => {});
            this.outputCtx = null;
          }
          this.onInputLevel(0);
          this.onOutputLevel(0);
        }
      }

      function App() {
        const [apiKey, setApiKey] = useState('');
        const [tempKeyInput, setTempKeyInput] = useState('');
        const [timeline, setTimeline] = useState([]);
        const [status, setStatus] = useState('idle');
        const [textInput, setTextInput] = useState('');
        const [inputLevel, setInputLevel] = useState(0);
        const [outputLevel, setOutputLevel] = useState(0);
        const [bpReadings, setBpReadings] = useState([]);

        const sessionRef = useRef(null);
        const audioManagerRef = useRef(null);
        const sessionTimerRef = useRef(null);
        const chatEndRef = useRef(null);
        const intentionalCloseRef = useRef(false);
        const currentUserMsgIdRef = useRef(null);
        const currentZoeMsgIdRef = useRef(null);

        useEffect(() => {
          chatEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
        }, [timeline]);

        const appendTimelineItem = useCallback((item) => {
          setTimeline((prev) => [...prev, item]);
        }, []);

        const updateOrAppendStreamingTranscript = useCallback((role, chunk) => {
          if (!chunk) return;
          const ref = role === 'user' ? currentUserMsgIdRef : currentZoeMsgIdRef;
          setTimeline((prev) => {
            const activeId = ref.current;
            if (activeId) {
              const idx = prev.findIndex((i) => i.id === activeId);
              if (idx !== -1 && prev[idx].type === 'message') {
                const updated = [...prev];
                updated[idx] = {
                  ...prev[idx],
                  text: prev[idx].text + chunk,
                  isPartial: true,
                };
                return updated;
              }
            }
            const newId = `${role}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
            ref.current = newId;
            return [
              ...prev,
              {
                id: newId,
                type: 'message',
                role,
                text: chunk,
                isPartial: true,
                timestamp: formatArmenianTime(),
              },
            ];
          });
        }, []);

        const finalizeStreamingTranscript = useCallback((role) => {
          const ref = role === 'user' ? currentUserMsgIdRef : currentZoeMsgIdRef;
          const activeId = ref.current;
          if (!activeId) return;
          setTimeline((prev) =>
            prev.map((item) =>
              item.id === activeId && item.type === 'message'
                ? { ...item, isPartial: false, text: item.text.trim() }
                : item
            )
          );
          ref.current = null;
        }, []);

        const cleanupSession = useCallback(() => {
          if (sessionTimerRef.current) {
            window.clearTimeout(sessionTimerRef.current);
            sessionTimerRef.current = null;
          }
          if (audioManagerRef.current) {
            audioManagerRef.current.stopAll();
            audioManagerRef.current = null;
          }
          if (sessionRef.current) {
            try {
              sessionRef.current.close();
            } catch {}
            sessionRef.current = null;
          }
          currentUserMsgIdRef.current = null;
          currentZoeMsgIdRef.current = null;
        }, []);

        const handleToolCall = useCallback(
          (functionCalls) => {
            if (!sessionRef.current) return;
            const functionResponses = functionCalls.map((call) => {
              const args = call.args || {};
              const now = formatArmenianTime();

              if (call.name === 'log_bp_reading') {
                const systolic = Number(args.systolic);
                const diastolic = Number(args.diastolic);
                const pulse = args.pulse !== undefined ? Number(args.pulse) : undefined;
                const symptoms = args.symptoms ? String(args.symptoms) : undefined;

                let bpStatus = 'target';
                if (systolic >= 160 || diastolic >= 100) bpStatus = 'alert';
                else if (systolic >= 130 || diastolic >= 80) bpStatus = 'elevated';

                const newReading = {
                  id: `bp-${Date.now()}-${Math.random().toString(36).slice(2, 5)}`,
                  systolic,
                  diastolic,
                  pulse,
                  symptoms,
                  timestamp: now,
                  status: bpStatus,
                };
                setBpReadings((prev) => [...prev, newReading]);
                appendTimelineItem({ id: newReading.id, type: 'bp_card', data: newReading });
                return { id: call.id, name: call.name, response: { status: 'logged' } };
              }

              if (call.name === 'alert_care_team') {
                const urgency = args.urgency === 'emergency' ? 'emergency' : 'same_day';
                const reason = String(
                  args.reason || 'Արյան ճնշման կամ ինքնազգացողության ահազանգ'
                );
                const alertId = `alert-${Date.now()}`;
                appendTimelineItem({
                  id: alertId,
                  type: 'alert_card',
                  data: { id: alertId, urgency, reason, timestamp: now },
                });
                return { id: call.id, name: call.name, response: { status: 'sent_demo' } };
              }

              if (call.name === 'set_reminder') {
                const medicine = String(args.medicine || 'Երեկոյան դեղահաբեր');
                const time = String(args.time || '20:00');
                const remId = `rem-${Date.now()}`;
                appendTimelineItem({
                  id: remId,
                  type: 'reminder_card',
                  data: { id: remId, medicine, time, timestamp: now },
                });
                return { id: call.id, name: call.name, response: { status: 'set_demo' } };
              }

              if (call.name === 'show_ldl_progress') {
                const ldlId = `ldl-${Date.now()}`;
                appendTimelineItem({
                  id: ldlId,
                  type: 'ldl_card',
                  data: { id: ldlId, timestamp: now },
                });
                return { id: call.id, name: call.name, response: { status: 'shown' } };
              }

              return { id: call.id, name: call.name, response: { status: 'unknown_tool' } };
            });

            try {
              sessionRef.current.sendToolResponse({ functionResponses });
            } catch (err) {
              console.error('Tool response error:', err);
            }
          },
          [appendTimelineItem]
        );

        const startLiveSession = useCallback(
          async (options) => {
            if (!apiKey) {
              appendTimelineItem({
                id: `sys-key-${Date.now()}`,
                type: 'system_notice',
                variant: 'info',

              onSpeakingStateChange: (speaking) => {
                setStatus((prev) => {

const WORKLET_CODE = `
class PcmCaptureProcessor extends AudioWorkletProcessor {
  constructor() {
    super();
    this.buffer = new Float32Array(2048);
    this.offset = 0;
  }
  process(inputs) {
    const input = inputs[0];
    if (!input || !input[0]) return true;
    const channel = input[0];
    for (let i = 0; i < channel.length; i++) {
      this.buffer[this.offset++] = channel[i];
      if (this.offset >= this.buffer.length) {
        this.port.postMessage(this.buffer.slice(0));
        this.offset = 0;
      }
    }
    return true;
  }
}
registerProcessor('pcm-capture-processor', PcmCaptureProcessor);
`;

export class LiveAudioManager {
  private inputCtx: AudioContext | null = null;
  private outputCtx: AudioContext | null = null;
  private micStream: MediaStream | null = null;
  private workletNode: AudioWorkletNode | null = null;
  private fallbackProcessor: ScriptProcessorNode | null = null;
  private sourceNode: MediaStreamAudioSourceNode | null = null;

  private activeSources: Set<AudioBufferSourceNode> = new Set();
  private nextStartTime = 0;

  private onInputLevel: (level: number) => void;
  private onOutputLevel: (level: number) => void;
  private onSpeakingStateChange: (isSpeaking: boolean) => void;

  constructor(callbacks: {
    onInputLevel: (level: number) => void;
    onOutputLevel: (level: number) => void;
    onSpeakingStateChange: (isSpeaking: boolean) => void;
  }) {
    this.onInputLevel = callbacks.onInputLevel;
    this.onOutputLevel = callbacks.onOutputLevel;
    this.onSpeakingStateChange = callbacks.onSpeakingStateChange;
  }

  private processFloat32Frame(
    inputData: Float32Array,
    onPcmChunk: (base64Pcm: string) => void
  ) {
    let sum = 0;
    for (let i = 0; i < inputData.length; i++) {
      sum += inputData[i] * inputData[i];
    }
    const rms = Math.sqrt(sum / inputData.length);
    this.onInputLevel(Math.min(1, rms * 5));

    const pcm16 = new Int16Array(inputData.length);
    for (let i = 0; i < inputData.length; i++) {
      const s = Math.max(-1, Math.min(1, inputData[i]));
      pcm16[i] = s < 0 ? s * 0x8000 : s * 0x7fff;
    }

    const uint8 = new Uint8Array(pcm16.buffer);
    let binary = '';
    for (let i = 0; i < uint8.byteLength; i++) {
      binary += String.fromCharCode(uint8[i]);
    }
    onPcmChunk(btoa(binary));
  }

  async startMicrophone(onPcmChunk: (base64Pcm: string) => void): Promise<void> {
    this.micStream = await navigator.mediaDevices.getUserMedia({
      audio: {
        sampleRate: 16000,
        channelCount: 1,
        echoCancellation: true,
        noiseSuppression: true,
        autoGainControl: true,
      },
    });

    const AudioCtx =
      window.AudioContext ||
      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;

    this.inputCtx = new AudioCtx({ sampleRate: 16000 });
    this.outputCtx = new AudioCtx({ sampleRate: 24000 });

    if (this.inputCtx.state === 'suspended') {
      await this.inputCtx.resume();
    }
    if (this.outputCtx.state === 'suspended') {
      await this.outputCtx.resume();
    }

    this.sourceNode = this.inputCtx.createMediaStreamSource(this.micStream);

    try {
      const blob = new Blob([WORKLET_CODE], { type: 'application/javascript' });
      const workletUrl = URL.createObjectURL(blob);
      await this.inputCtx.audioWorklet.addModule(workletUrl);
      URL.revokeObjectURL(workletUrl);

      this.workletNode = new AudioWorkletNode(this.inputCtx, 'pcm-capture-processor');
      this.workletNode.port.onmessage = (event: MessageEvent<Float32Array>) => {
        this.processFloat32Frame(event.data, onPcmChunk);
      };

      this.sourceNode.connect(this.workletNode);
      this.workletNode.connect(this.inputCtx.destination);
    } catch {
      this.fallbackProcessor = this.inputCtx.createScriptProcessor(2048, 1, 1);
      this.fallbackProcessor.onaudioprocess = (event) => {
        const inputData = event.inputBuffer.getChannelData(0);
        this.processFloat32Frame(inputData, onPcmChunk);
      };
      this.sourceNode.connect(this.fallbackProcessor);
      this.fallbackProcessor.connect(this.inputCtx.destination);
    }
  }

  enqueueOutputPcm(base64Data: string): void {
    if (!this.outputCtx) return;

    if (this.outputCtx.state === 'suspended') {
      this.outputCtx.resume();
    }

    const binaryString = atob(base64Data);
    const byteLength = binaryString.length;
    const bytes = new Uint8Array(byteLength);
    for (let i = 0; i < byteLength; i++) {
      bytes[i] = binaryString.charCodeAt(i);
    }

    const int16 = new Int16Array(bytes.buffer);
    const float32 = new Float32Array(int16.length);
    let sum = 0;
    for (let i = 0; i < int16.length; i++) {
      const sample = int16[i] / 32768.0;
      float32[i] = sample;
      sum += sample * sample;
    }

    const rms = Math.sqrt(sum / Math.max(1, float32.length));
    this.onOutputLevel(Math.min(1, rms * 4.5));

    const audioBuffer = this.outputCtx.createBuffer(1, float32.length, 24000);
    audioBuffer.getChannelData(0).set(float32);

    const source = this.outputCtx.createBufferSource();
    source.buffer = audioBuffer;
    source.connect(this.outputCtx.destination);

    const now = this.outputCtx.currentTime;
    if (this.nextStartTime < now) {
      this.nextStartTime = now + 0.02;
    }

    source.start(this.nextStartTime);
    this.nextStartTime += audioBuffer.duration;

    this.activeSources.add(source);
    this.onSpeakingStateChange(true);

    source.onended = () => {
      this.activeSources.delete(source);
      if (this.activeSources.size === 0) {
        this.onOutputLevel(0);
        this.onSpeakingStateChange(false);
      }
    };
  }

  interruptOutput(): void {
    this.activeSources.forEach((src) => {
      try {
        src.stop();
      } catch {
        // Ignore already stopped sources
      }
    });
    this.activeSources.clear();
    if (this.outputCtx) {
      this.nextStartTime = this.outputCtx.currentTime;
    }
    this.onOutputLevel(0);
    this.onSpeakingStateChange(false);
  }

  stopAll(): void {
    this.interruptOutput();

    if (this.workletNode) {
      this.workletNode.port.onmessage = null;
      this.workletNode.disconnect();
      this.workletNode = null;
    }
    if (this.fallbackProcessor) {
      this.fallbackProcessor.disconnect();
      this.fallbackProcessor.onaudioprocess = null;
      this.fallbackProcessor = null;
    }
    if (this.sourceNode) {
      this.sourceNode.disconnect();
      this.sourceNode = null;
    }
    if (this.micStream) {
      this.micStream.getTracks().forEach((t) => t.stop());
      this.micStream = null;
    }
    if (this.inputCtx) {
      this.inputCtx.close().catch(() => {});
      this.inputCtx = null;
    }
    if (this.outputCtx) {
      this.outputCtx.close().catch(() => {});
      this.outputCtx = null;
    }
    this.onInputLevel(0);
    this.onOutputLevel(0);
  }
}
